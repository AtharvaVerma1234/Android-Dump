// -------------------------------------------------------------------------
// Big-Endian Helpers for Reading UDP Payload Bytes from ethernetPacket*
// -------------------------------------------------------------------------

word ReadUdpBeWord(ethernetPacket* pckt, dword offset)
{
  return ((word)pckt.udp.byte(offset) << 8) | (word)pckt.udp.byte(offset + 1);
}

dword ReadUdpBe24(ethernetPacket* pckt, dword offset)
{
  return ((dword)pckt.udp.byte(offset)     << 16) |
         ((dword)pckt.udp.byte(offset + 1) << 8)  |
          (dword)pckt.udp.byte(offset + 2);
}

dword ReadUdpBeDWord(ethernetPacket* pckt, dword offset)
{
  return ((dword)pckt.udp.byte(offset)     << 24) |
         ((dword)pckt.udp.byte(offset + 1) << 16) |
         ((dword)pckt.udp.byte(offset + 2) << 8)  |
          (dword)pckt.udp.byte(offset + 3);
}

// -------------------------------------------------------------------------
// Main SOME/IP-SD Packet Printer
// -------------------------------------------------------------------------

void print_sd_packet_details(ethernetPacket* pckt)
{
  dword udpPayloadLen;
  word  serviceId;
  word  methodId;
  word  clientId;
  word  sessionId;
  byte  flags;
  byte  rebootFlag;
  byte  unicastFlag;

  dword entriesLen;
  dword entriesEnd;
  dword offset;
  int   entryIdx;

  byte  entryType;
  byte  idx1stOpt;
  byte  idx2ndOpt;
  byte  numOpt1;
  byte  numOpt2;
  word  eSrvId;
  word  eInstId;
  byte  eMajVer;
  dword eTtl;
  dword eMinVer;
  byte  b13;
  byte  initFlag;
  byte  counter;
  word  evgId;
  char  typeLabel[32];

  dword optionsLen;
  dword optOffset;
  dword optionsEnd;
  int   optIdx;
  word  optLen;
  byte  optType;
  dword totalOptBytes;
  byte  l4Proto;
  word  portNum;

  if (pckt == null || !pckt.udp.IsAvailable())
  {
    write("[SOME/IP-SD] Invalid or non-UDP packet.");
    return;
  }

  udpPayloadLen = pckt.udp.Length - 8;
  if (udpPayloadLen < 28) // 16B SOME/IP Header + 4B SD Header + 4B EntriesLen + 4B OptionsLen
  {
    write("[SOME/IP-SD] Packet too short (%d bytes).", udpPayloadLen);
    return;
  }

  serviceId = ReadUdpBeWord(pckt, 0);
  methodId  = ReadUdpBeWord(pckt, 2);

  if (serviceId != 0xFFFF || methodId != 0x8100)
  {
    write("[SOME/IP-SD] Not an SD packet (ServiceID=0x%04X, MethodID=0x%04X).", serviceId, methodId);
    return;
  }

  clientId    = ReadUdpBeWord(pckt, 8);
  sessionId   = ReadUdpBeWord(pckt, 10);
  flags       = pckt.udp.byte(16);
  rebootFlag  = (flags >> 7) & 0x01;
  unicastFlag = (flags >> 6) & 0x01;
  entriesLen  = ReadUdpBeDWord(pckt, 20);
  entriesEnd  = 24 + entriesLen;

  if (entriesEnd + 4 > udpPayloadLen)
  {
    write("[SOME/IP-SD] Malformed packet: EntriesLength (%u) exceeds payload.", entriesLen);
    return;
  }

  write("================ SOME/IP-SD PACKET ================");
  write("Header  | ClientID: 0x%04X | SessionID: 0x%04X | Flags: 0x%02X (Reboot=%d, Unicast=%d)",
        clientId, sessionId, flags, rebootFlag, unicastFlag);
  write("Lengths | Entries: %u bytes (%u entries) | UDP Payload: %u bytes",
        entriesLen, entriesLen / 16, udpPayloadLen);

  // --- Parse Entries Array ---
  offset   = 24;
  entryIdx = 0;

  while (offset + 16 <= entriesEnd)
  {
    entryType = pckt.udp.byte(offset);
    idx1stOpt = pckt.udp.byte(offset + 1);
    idx2ndOpt = pckt.udp.byte(offset + 2);
    numOpt1   = (pckt.udp.byte(offset + 3) >> 4) & 0x0F;
    numOpt2   = pckt.udp.byte(offset + 3) & 0x0F;
    eSrvId    = ReadUdpBeWord(pckt, offset + 4);
    eInstId   = ReadUdpBeWord(pckt, offset + 6);
    eMajVer   = pckt.udp.byte(offset + 8);
    eTtl      = ReadUdpBe24(pckt, offset + 9);

    switch (entryType)
    {
      case 0x00: snprintf(typeLabel, elcount(typeLabel), "FindService"); break;
      case 0x01: snprintf(typeLabel, elcount(typeLabel), (eTtl == 0) ? "StopOfferService" : "OfferService"); break;
      case 0x06: snprintf(typeLabel, elcount(typeLabel), (eTtl == 0) ? "StopSubscribeEvg" : "SubscribeEventgroup"); break;
      case 0x07: snprintf(typeLabel, elcount(typeLabel), (eTtl == 0) ? "SubscribeEvgNack"  : "SubscribeEventgroupAck"); break;
      default:   snprintf(typeLabel, elcount(typeLabel), "Unknown(0x%02X)", entryType); break;
    }

    if (entryType == 0x00 || entryType == 0x01)
    {
      eMinVer = ReadUdpBeDWord(pckt, offset + 12);
      write("  Entry[%d] %-22s | Srv: 0x%04X | Inst: 0x%04X | Ver: %u.%u | TTL: %u s | Opts: [Idx:%u,%u Num:%u,%u]",
            entryIdx, typeLabel, eSrvId, eInstId, eMajVer, eMinVer, eTtl, idx1stOpt, idx2ndOpt, numOpt1, numOpt2);
    }
    else
    {
      b13      = pckt.udp.byte(offset + 13);
      initFlag = (b13 >> 7) & 0x01;
      counter  = b13 & 0x0F;
      evgId    = ReadUdpBeWord(pckt, offset + 14);
      write("  Entry[%d] %-22s | Srv: 0x%04X | Inst: 0x%04X | EvgID: 0x%04X | MajVer: %u | TTL: %u s | InitReq: %u | Cnt: %u | Opts: [Idx:%u,%u Num:%u,%u]",
            entryIdx, typeLabel, eSrvId, eInstId, evgId, eMajVer, eTtl, initFlag, counter, idx1stOpt, idx2ndOpt, numOpt1, numOpt2);
    }

    offset += 16;
    entryIdx++;
  }

  // --- Parse Options Array (if present) ---
  optionsLen = ReadUdpBeDWord(pckt, entriesEnd);
  optOffset  = entriesEnd + 4;
  optionsEnd = optOffset + optionsLen;
  optIdx     = 0;

  if (optionsLen == 0)
  {
    write("Options | None (0 bytes)");
  }
  else if (optionsEnd > udpPayloadLen)
  {
    write("[SOME/IP-SD] Malformed packet: OptionsLength (%u) exceeds payload.", optionsLen);
  }
  else
  {
    write("Options | %u bytes:", optionsLen);
    while (optOffset + 3 <= optionsEnd)
    {
      optLen        = ReadUdpBeWord(pckt, optOffset);
      optType       = pckt.udp.byte(optOffset + 2);
      totalOptBytes = 3 + optLen; // AUTOSAR Option Length excludes Length(2B) + Type(1B)

      if (optOffset + totalOptBytes > optionsEnd) break;

      if ((optType == 0x04 || optType == 0x14) && optLen == 0x0009)
      {
        l4Proto = pckt.udp.byte(optOffset + 9);
        portNum = ReadUdpBeWord(pckt, optOffset + 10);
        write("  Option[%d] %-14s | IP: %u.%u.%u.%u | Proto: %s (0x%02X) | Port: %u",
              optIdx,
              (optType == 0x04) ? "IPv4 Endpoint" : "IPv4 Multicast",
              pckt.udp.byte(optOffset + 4), pckt.udp.byte(optOffset + 5),
              pckt.udp.byte(optOffset + 6), pckt.udp.byte(optOffset + 7),
              (l4Proto == 0x11) ? "UDP" : ((l4Proto == 0x06) ? "TCP" : "OTHER"),
              l4Proto, portNum);
      }
      else
      {
        write("  Option[%d] Type: 0x%02X | Length: %u bytes", optIdx, optType, optLen);
      }

      optOffset += totalOptBytes;
      optIdx++;
    }
  }

  write("===================================================");
}
